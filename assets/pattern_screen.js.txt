function drawChart(){if(!chartDoc||!selected)return;const s=selected,all=chartDoc.bars,offset=Math.max(0,all.length-range),bars=all.slice(offset),N=bars.length,W=Math.max(800,N*7+100),H=470,L=54,R=W-65,T=68,B=330,V=405,step=(R-L)/(N||1);const x=i=>L+(i-offset+.5)*step;
 const ma=(i,n)=>i+1<n?null:all.slice(i-n+1,i+1).reduce((a,b)=>a+b.close,0)/n;
 const overlays=$('#psOverlay').checked;const included=bars.flatMap(b=>[b.high,b.low]);
 const ns=[50,150,200].filter(n=>$('#psMA'+n).checked);for(const n of ns)for(let i=offset;i<all.length;i++){const v=ma(i,n);if(v!==null)included.push(v);}
